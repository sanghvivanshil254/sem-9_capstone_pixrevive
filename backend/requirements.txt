# PixRevive backend — runtime dependencies.
#
# Install into the venv the backend runs from:
#     python -m pip install -r backend/requirements.txt
#
# NOTE on Torch: the pinned CUDA build is not on PyPI. Install it first from the
# PyTorch index, then run this file (pip will keep the already-satisfied build):
#     python -m pip install torch torchvision --index-url https://download.pytorch.org/whl/cu126
# Omit the index URL for a CPU-only install; the pipeline falls back to CPU
# automatically when torch.cuda.is_available() is False.
